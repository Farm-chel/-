    /// <summary>
    /// Логика взаимодействия для FirstWindow.xaml
    /// </summary>
    public partial class FirstWindow : Window
    {
        public FirstWindow()
        {
            InitializeComponent();
            LoadUsers();
        }
        private void LoadUsers()
        {
            using (var db = new хранительПРОEntities1())
            {
                UsersDataGrid.ItemsSource = db.Users.ToList();
            }
        }


        private void ApplyFilters()
        {
            using (var db = new хранительПРОEntities1())
            {
                var users = db.Users.AsQueryable();

                // Фильтр по email
                if (!string.IsNullOrWhiteSpace(EmailFilterTextBox.Text))
                {
                    users = users.Where(u => u.email.Contains(EmailFilterTextBox.Text));
                }

                // Фильтр по роли
                if (RoleFilterComboBox.SelectedItem is ComboBoxItem selectedRole &&
                    selectedRole.Content.ToString() != "Все роли")
                {
                    users = users.Where(u => u.Роль == selectedRole.Content.ToString());
                }

               // UsersDataGrid.ItemsSource = users.ToList();
            }
        }

        private void FilterTextBox_TextChanged(object sender, TextChangedEventArgs e)
        {
            ApplyFilters();
        }

        private void FilterComboBox_SelectionChanged(object sender, SelectionChangedEventArgs e)
        {
            ApplyFilters();
        }

        private void ResetFilters_Click(object sender, RoutedEventArgs e)
        {
            EmailFilterTextBox.Text = string.Empty;
            RoleFilterComboBox.SelectedIndex = 0;
            LoadUsers();
        }

        private void Button_Click(object sender, RoutedEventArgs e)
        {
            ViewRequestsWindow viewRequestsWindow = new ViewRequestsWindow();
            viewRequestsWindow.Show();
            this.Close();
        }

        private void RegisterButton_Click(object sender, RoutedEventArgs e)
        {
            RegistrationWindow registrationWindow = new RegistrationWindow();
            registrationWindow.Show();
            this.Close();
        }

        private void LoginButton_Click(object sender, RoutedEventArgs e)
        {
            LoginWindow loginWindow = new LoginWindow();
            loginWindow.Show();
            this.Close();
        }

        private void CreateButton_Click(object sender, RoutedEventArgs e)
        {
            MainWindow window = new MainWindow();
            window.Show();
            this.Close();
        }
    }

    {
        public LoginWindow()
        {
            InitializeComponent();
        }

        private void LoginButton_Click(object sender, RoutedEventArgs e)
        {
            string email = EmailTextBox.Text;
            string password = PasswordBox.Password;

            using (var db = new хранительПРОEntities1())
            {
                // Проверка наличия пользователя с введенными данными
                var user = db.Users.FirstOrDefault(u => u.email == email && u.Пароль == password);

                if (user != null)
                {
                    // Успешная авторизация
                    MessageBox.Show("Авторизация успешна!");
                    // Здесь можно открыть главное окно приложения
                    MainWindow mainWindow = new MainWindow();
                    mainWindow.Show();
                    this.Close(); // Закрываем окно авторизации
                }
                else
                {
                    // Ошибка авторизации
                    MessageBox.Show("Неверный email или пароль.");
                }
            }
        }

        {

    /// <summary>
    /// Логика взаимодействия для RegistrationWindow.xaml
    /// </summary>
    public partial class RegistrationWindow : Window
    {
        public RegistrationWindow()
        {
            InitializeComponent();
        }

        private void RegisterButton_Click(object sender, RoutedEventArgs e)
        {
            string email = EmailTextBox.Text;
            string password = PasswordBox.Password;
            string confirmPassword = ConfirmPasswordBox.Password;

            if (password != confirmPassword)
            {
                MessageBox.Show("Пароли не совпадают.");
                return;
            }

            using (var db = new хранительПРОEntities1())
            {
                // Проверка на существование пользователя с таким email
                if (db.Users.Any(u => u.email == email))
                {
                    MessageBox.Show("Пользователь с таким email уже существует.");
                    return;
                }

                // Создание нового пользователя
                Users newUser = new Users
                {
                    email = email,
                    Пароль = password, // Здесь можно добавить хеширование пароля
                    Роль = "User " // Установите роль по умолчанию
                };

                db.Users.Add(newUser);
                db.SaveChanges(); // Сохраняем изменения в базе данных
            }

            MessageBox.Show("Регистрация успешна! Теперь вы можете войти.");
            this.Close(); // Закрываем окно регистрации
        }

        private void BackButton_Click(object sender, RoutedEventArgs e)
        {
            FirstWindow firstWindow = new FirstWindow();
            firstWindow.Show();
            this.Close();
        }
    }
}
